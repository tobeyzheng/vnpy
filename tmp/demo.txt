class Strategy(StrategyBase):

    def initialize(self): # 初始化
        declare_strategy_type(AlgoStrategyType.SECURITY)
        self.trigger_symbols()
        self.custom_indicator()
        self.global_variables()

    def trigger_symbols(self):  # 定义驱动标的
        self.驱动标的1 = declare_trig_symbol()

    def global_variables(self):  # 定义全局变量
        self.提醒价格 = show_variable(100, GlobalType.INT)

    def custom_indicator(self):
        # 定义麦语言自编指标，参见手册 register_indicator()
        # 定义Python自编指标，参见手册 register_indicator_Python()
        pass


    # 策略的主函数。驱动标的的行情更新，或者到达指定时间，会触发handle_data()函数
    def handle_data(self):
        self.condition_alert_invoke()  # 示例：定义一个逻辑分支，支持定义多个逻辑分支


    # 示例：价格大于100元，则推送消息
    def condition_alert_invoke(self):
        if self.condition_alert():
            self.action_alert()
        else:
            pass


    # 示例：判断价格是否大于100元
    def condition_alert(self):
        v_price = current_price(symbol=self.驱动标的1, price_type=THType.ALL)
        if v_price > self.提醒价格:
            return True
        else:
            return False


    # 示例：推送消息
    def action_alert(self):
        alert(content="价格大于100元")
